VAR NbJours = CALCULATE ( DISTINCTCOUNT ( 'Stock Carburant 11'[Date] ) )

VAR Base1 =
    ADDCOLUMNS (
        VALUES ( 'Stock Carburant 11'[Type] ),
        "cCouleur",
            SWITCH ( 'Stock Carburant 11'[Type],
                "GASOIL", "#F5B800", "GASOIL+", "#10B981",
                "SUPER", "#3B82F6", "SUPER+", "#A78BFA", "#94A3B8" ),
        "cComptable",
            VAR d = CALCULATE ( MAX ( 'Stock Carburant 11'[Date] ) )
            RETURN CALCULATE ( SUM ( 'Stock Carburant 11'[Stock_L] ), 'Stock Carburant 11'[Date] = d ),
        "cReel",    CALCULATE ( MAX ( 'Stock Carburant 11'[Stock_Reel_Mois] ) ),
        "cLitres",  CALCULATE ( SUM ( 'Stock Carburant 11'[Sorties_L] ) ),
        "cDernier", CALCULATE ( MAX ( 'Stock Carburant 11'[Date] ) )
    )
VAR Base2 =
    ADDCOLUMNS (
        Base1,
        "cStock", IF ( NOT ISBLANK ( [cReel] ), [cReel], [cComptable] ),
        "cMoy",   DIVIDE ( [cLitres], NbJours )
    )
VAR Base =
    ADDCOLUMNS ( Base2, "cAut", DIVIDE ( [cStock], [cMoy], 9999 ) )

VAR TotLitres = SUMX ( Base, [cLitres] )
VAR TotStock  = SUMX ( Base, [cStock] )
VAR NbCrit    = COUNTROWS ( FILTER ( Base, [cAut] < 3 ) )

VAR Base4 =
    ADDCOLUMNS (
        Base,
        "cId",    RANKX ( Base, [Type], , ASC, DENSE ),
        "cEc",    IF ( ISBLANK ( [cReel] ), BLANK (), ROUND ( [cReel] - [cComptable], 2 ) ),
        "cCoul",  IF ( [cAut] < 3, "#FF6B6B", IF ( [cAut] < 7, "#FFB020", "#34D399" ) ),
        "cBgSt",  IF ( [cAut] < 3, "rgba(255,107,107,.20)", IF ( [cAut] < 7, "rgba(255,176,32,.20)", "rgba(52,211,153,.20)" ) ),
        "cEtat",  IF ( [cAut] < 3, "CRITIQUE", IF ( [cAut] < 7, "A SURVEILLER", "SECURISE" ) ),
        "cPct",   MAX ( MIN ( DIVIDE ( [cAut], 10 ), 1 ), 0.07 ) * 100,
        "cTxtJ",  IF ( [cAut] >= 99, "99+", IF ( [cAut] < 1, FORMAT ( ROUND ( [cAut] * 24, 0 ), "0" ), FORMAT ( [cAut], "0.0", "fr-FR" ) ) ),
        "cUnit",  IF ( [cAut] < 1, "HEURES", "JOURS" ),
        "cFirst", IF ( [cAut] = MINX ( Base, [cAut] ), 1, 0 ),
        "cPart",  FORMAT ( DIVIDE ( [cLitres], TotLitres ), "0.0%", "fr-FR" ),
        "cRupt",  IF ( [cAut] >= 99, "&gt; 99 j", FORMAT ( [cDernier] + INT ( [cAut] ), "dd/mm", "fr-FR" ) )
    )
